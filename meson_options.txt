# MisraStdC feature flags.
#
# The library is split into a small mandatory foundation (Sys, Mutex, Log,
# Memory, Allocator core + Page + Heap) and a set of optional features that
# can be turned off if the consumer does not need them. Disabling a feature
# drops its source files from the build, shrinking the static library and
# the final executable.
#
# Dependencies are resolved automatically: enabling a feature whose
# implementation needs another feature transparently enables that other
# feature too. So `-Dgraph=true -Dvec=false` is not a conflict: `vec` is
# automatically turned back on because `Graph` links into `vec_*` runtime
# helpers.
#
# Each enabled feature also defines `MISRA_HAVE_<FEATURE>=1` at the
# compile-time level, so downstream consumers can `#ifdef MISRA_HAVE_VEC`
# in their own code if they need to compile against a partial build.

# --- Optional allocator backends --------------------------------------------
option('alloc_arena',  type: 'boolean', value: true,  description: 'Build ArenaAllocator (bump allocator).')
option('alloc_slab',   type: 'boolean', value: true,  description: 'Build SlabAllocator (growable fixed-slot pool).')
option('alloc_budget', type: 'boolean', value: true,  description: 'Build BudgetAllocator (caller-buffer bounded pool, no growth).')
option('alloc_stats',  type: 'boolean', value: true,  description: 'Track per-Allocator memory-pressure stats (byte counts, alloc/realloc/free counters, peak usage).')
option('alloc_debug',  type: 'boolean', value: true,  description: 'Build DebugAllocator wrapper (GPA-style leak / double-free / overflow / stack-trace tracking). Requires map and sys_backtrace.')
option('default_alloc_debug', type: 'boolean', value: false, description: 'Alias DefaultAllocator to DebugAllocator (per-instance, no globals). Drop-in ASan/MSan-style leak + double-free + overflow detection without changing call sites. Requires alloc_debug.')

# Note: Vec, Str, Io are part of the mandatory foundation because
# `LOG_FATAL` formats its message through them. They are not exposed as
# feature options. The remaining containers below are optional.

# --- Containers -------------------------------------------------------------
option('bitvec', type: 'boolean', value: true, description: 'Build BitVec packed bit container.')
option('list',   type: 'boolean', value: true, description: 'Build List doubly-linked list container.')
option('map',    type: 'boolean', value: true, description: 'Build Map hash-map container.')
option('graph',  type: 'boolean', value: true, description: 'Build Graph directed-graph container.')
option('int',    type: 'boolean', value: true, description: 'Build Int arbitrary-precision integer. Requires bitvec.')
option('float',  type: 'boolean', value: true, description: 'Build Float arbitrary-precision decimal. Requires int.')

# --- Std subsystems ---------------------------------------------------------
option('file', type: 'boolean', value: true, description: 'Build File helpers (ReadCompleteFile, ...).')
option('iter', type: 'boolean', value: true, description: 'Build Iter utility helpers.')

# --- Sys subsystems ---------------------------------------------------------
option('sys_dir',    type: 'boolean', value: true, description: 'Build directory-handling syscalls (DirContents, ...).')
option('sys_proc',   type: 'boolean', value: true, description: 'Build process-spawning syscalls (ProcCreate, ...).')
option('sys_socket',    type: 'boolean', value: true, description: 'Build TCP/UDP socket abstraction (Listener, Socket, SocketPoll, ...). POSIX only at present.')
option('sys_procmaps',  type: 'boolean', value: true, description: 'Build /proc/self/maps parser (Linux only). Used by the in-tree dladdr replacement.')
option('sys_symresolve',type: 'boolean', value: true, description: 'Build SymbolResolver (in-tree dladdr replacement). Requires parser_elf + sys_procmaps.')
option('sys_backtrace', type: 'boolean', value: true, description: 'Build Sys/Backtrace (in-tree frame-pointer unwinder + symbol-resolving formatter). Requires sys_symresolve.')

# --- Parsers ----------------------------------------------------------------
option('parser_json',     type: 'boolean', value: true, description: 'Build JSON reader/writer.')
option('parser_kvconfig', type: 'boolean', value: true, description: 'Build key-value config parser. Requires map, str.')
option('parser_http',     type: 'boolean', value: true, description: 'Build HTTP/1.1 request + response parser/serializer.')
option('parser_elf',      type: 'boolean', value: true, description: 'Build ELF64 parser (header, sections, symbol tables). Used by the in-tree dladdr replacement.')
option('parser_dwarf',    type: 'boolean', value: true, description: 'Build DWARF parsers: .debug_line (IP → file:line) and .eh_frame CFI (frame-pointer-less stack unwinding). Requires parser_elf.')
option('parser_pe',       type: 'boolean', value: true, description: 'Build PE/COFF parser. Used by the Windows backtrace path to locate the CodeView (PDB) reference in a PE binary.')
option('parser_pdb',      type: 'boolean', value: true, description: 'Build PDB reader (MSF container + DBI/Publics streams). Used by the Windows backtrace path to resolve symbols without dbghelp. Requires parser_pe.')
option('parser_macho',    type: 'boolean', value: true, description: 'Build Mach-O parser. Used by the macOS backtrace path to walk loaded images.')
